cmake_minimum_required(VERSION 3.16)
project(jetbot_orb_slam2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d features2d)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)

set(ORB_SLAM2_SOURCE "" CACHE PATH "Path to patched ORB-SLAM2 sources")
add_subdirectory("${ORB_SLAM2_SOURCE}/Thirdparty/DBoW2" dbow2)
add_subdirectory("${ORB_SLAM2_SOURCE}/Thirdparty/g2o" g2o)
file(GLOB orb_sources CONFIGURE_DEPENDS "${ORB_SLAM2_SOURCE}/src/*.cc")
list(FILTER orb_sources EXCLUDE REGEX "/(MapDrawer|Viewer)\\.cc$")
add_library(orb_slam2 SHARED ${orb_sources})
target_include_directories(orb_slam2 PUBLIC "${ORB_SLAM2_SOURCE}" "${ORB_SLAM2_SOURCE}/include")
target_compile_options(orb_slam2 PRIVATE -include unistd.h)
target_link_libraries(orb_slam2 PUBLIC ${OpenCV_LIBS} Eigen3::Eigen Threads::Threads DBoW2 g2o)
add_executable(jetbot_orb_replay replay.cc)
target_link_libraries(jetbot_orb_replay PRIVATE orb_slam2)
